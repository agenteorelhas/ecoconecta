O código que faz o site funcionar

// Impede o recarregamento e valida cada campo
form.addEventListener('submit', (e) => {
  e.preventDefault();          // não recarrega a página
  let tudo_ok = true;
  Object.entries(campos).forEach(([key, cfg]) => {
    if (!validarCampo(key, cfg)) tudo_ok = false;
  });
  if (!tudo_ok) return;        // foca no primeiro erro
  localStorage.removeItem(key); // limpa rascunho
});

// Salva rascunho enquanto o usuário digita
campo.addEventListener('input', () => {
  localStorage.setItem('form_' + key, campo.value);
});
